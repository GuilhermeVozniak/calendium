```ts
import type { Event, Thread } from '@calendium/shared';

import { getApiClient } from '@/lib/api';
import { calendarMock } from '@/lib/calendar-mock';
import { DEMO_MODE } from '@/lib/demo';
import { getMockThreads } from '@/lib/mail-mock';

export interface SearchData {
  threads: Thread[];
  events: Event[];
}

/** Unified search (GET /v1/search). Demo mode filters local mocks. */
export async function fetchSearch(q: string): Promise<SearchData> {
  try {
    return await getApiClient().search(q);
  } catch (err) {
    if (!DEMO_MODE) throw err;
    const needle = q.toLowerCase();
    const from = new Date();
    from.setDate(from.getDate() - 90);
    const to = new Date();
    to.setDate(to.getDate() + 90);
    return {
      threads: getMockThreads({ q }).items.slice(0, 6),
      events: calendarMock
        .listEvents(from.toISOString(), to.toISOString())
        .filter((e: Event) => e.title.toLowerCase().includes(needle))
        .slice(0, 6),
    };
  }
}
```

(Adjust the two mock calls to the actual signatures at `mail-mock.ts:551` / `calendar-mock.ts:268` — e.g. `getMockThreads` may return `{ items, nextCursor }` or a plain array, and `listEvents` may take `Date`s.)

```ts
export interface SnoozePreset {
  label: string;
  until: Date;
}

/** Canonical snooze presets shared by inbox swipe/long-press and the thread reader. */
export function snoozePresets(now: Date = new Date()): SnoozePreset[] {
  const laterToday = new Date(now.getTime() + 3 * 60 * 60 * 1000);
  const tomorrow = new Date(now);
  tomorrow.setDate(tomorrow.getDate() + 1);
  tomorrow.setHours(8, 0, 0, 0);
  const nextWeek = new Date(now);
  nextWeek.setDate(nextWeek.getDate() + 7);
  nextWeek.setHours(8, 0, 0, 0);
  return [
    { label: 'Later today', until: laterToday },
    { label: 'Tomorrow 8 AM', until: tomorrow },
    { label: 'Next week', until: nextWeek },
  ];
}
```
